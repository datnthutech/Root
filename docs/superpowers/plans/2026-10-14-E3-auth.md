# E3 Xác thực và phân quyền Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Đăng ký (chờ duyệt) → duyệt → đăng nhập bằng JWT 15 phút + refresh cookie httpOnly; khoá sau 5 lần sai; phân quyền ADMIN > EDITOR > VIEWER ở API; màn hình đăng nhập, đăng ký, duyệt tài khoản; chưa đăng nhập thì chỉ thấy trang đăng nhập.

**Architecture:** Spring Security stateless; `JwtAuthFilter` đọc `Authorization: Bearer`; quyền kiểm bằng `@PreAuthorize("hasRole('EDITOR')")` với `RoleHierarchy`. Refresh token ngẫu nhiên, lưu SHA-256 trong DB, xoay vòng mỗi lần refresh. FE giữ access token trong bộ nhớ, tự refresh khi gặp 401.

**Tech Stack:** Spring Security 6.3, jjwt 0.12, BCrypt; React Router, Ant Design Form/Table.

**Spec:** `docs/superpowers/specs/2026-10-14-gia-pha-design.md`; tài liệu gốc mục "Phân quyền".

## Global Constraints

- JWT sống **15 phút**; refresh token **14 ngày**, cookie `refresh_token`: `HttpOnly; SameSite=Strict; Path=/api/v1/auth`; `Secure` bật bằng `app.cookie-secure=true` ở môi trường thật.
- Khoá tạm sau **5** lần sai mật khẩu liên tiếp (15 phút); tài khoản `LOCKED` do admin khoá thì khoá tới khi mở.
- Mã lỗi: `UNAUTHENTICATED`(401), `BAD_CREDENTIALS`(401), `ACCOUNT_PENDING`(403), `ACCOUNT_LOCKED`(423), `FORBIDDEN`(403), `USERNAME_TAKEN`(409) `[BỔ SUNG]`.
- Mật khẩu 8–72 ký tự; `username` 3–50 ký tự `[a-zA-Z0-9._-]`, lưu chữ thường.
- Không đăng nhập thì mọi `/api/v1/**` trừ `/auth/register|login|refresh|logout` trả 401.

## Review Focus

- Đăng nhập sai 5 lần rồi nhập **đúng** mật khẩu vẫn bị `ACCOUNT_LOCKED`; bộ đếm sai phải được lưu dù request trả lỗi (giao dịch không được rollback).
- Refresh token cũ đã dùng (xoay vòng) hoặc đã đăng xuất không dùng lại được.
- Tài khoản `PENDING` nhập sai mật khẩu nhận `BAD_CREDENTIALS`, không lộ trạng thái chờ duyệt.
- Admin không thể tự hạ quyền / tự khoá chính mình (tránh hệ thống không còn admin).
- Hai request refresh đồng thời ở FE chỉ gọi API một lần (single-flight).

```java
package vn.giapha.auth;

import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.Date;
import java.util.Optional;
import javax.crypto.SecretKey;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

@Service
public class JwtService {
    private final SecretKey key;
    private final long accessMinutes;

    public JwtService(@Value("${app.jwt.secret}") String secret, @Value("${app.jwt.access-minutes}") long accessMinutes) {
        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
        this.accessMinutes = accessMinutes;
    }

    public String issue(long userId, Role role) {
        Instant now = Instant.now();
        return Jwts.builder().subject(String.valueOf(userId)).claim("role", role.name())
            .issuedAt(Date.from(now)).expiration(Date.from(now.plus(accessMinutes, ChronoUnit.MINUTES)))
            .signWith(key).compact();
    }

    public Optional<AuthUser> parse(String token) {
        try {
            var c = Jwts.parser().verifyWith(key).build().parseSignedClaims(token).getPayload();
            return Optional.of(new AuthUser(Long.parseLong(c.getSubject()), Role.valueOf(c.get("role", String.class))));
        } catch (JwtException | IllegalArgumentException e) {
            return Optional.empty();
        }
    }

    public long accessSeconds() { return accessMinutes * 60; }
}
```

```java
package vn.giapha.auth;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.List;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
public class JwtAuthFilter extends OncePerRequestFilter {
    private final JwtService jwt;

    public JwtAuthFilter(JwtService jwt) { this.jwt = jwt; }

    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
            throws ServletException, IOException {
        String h = req.getHeader("Authorization");
        if (h != null && h.startsWith("Bearer ")) {
            jwt.parse(h.substring(7)).ifPresent(u -> SecurityContextHolder.getContext().setAuthentication(
                new UsernamePasswordAuthenticationToken(u, null, List.of(new SimpleGrantedAuthority("ROLE_" + u.role().name())))));
        }
        chain.doFilter(req, res);
    }
}
```

```java
package vn.giapha.auth;

import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.MediaType;
import org.springframework.security.access.hierarchicalroles.RoleHierarchy;
import org.springframework.security.access.hierarchicalroles.RoleHierarchyImpl;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.security.access.expression.method.DefaultMethodSecurityExpressionHandler;
import org.springframework.security.access.expression.method.MethodSecurityExpressionHandler;
import vn.giapha.common.ApiError;

@Configuration
@EnableMethodSecurity
public class SecurityConfig {

    @Bean
    SecurityFilterChain chain(HttpSecurity http, JwtAuthFilter jwtFilter, ObjectMapper om) throws Exception {
        http.csrf(c -> c.disable())
            .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(a -> a
                .requestMatchers("/api/v1/auth/register", "/api/v1/auth/login", "/api/v1/auth/refresh", "/api/v1/auth/logout",
                    "/actuator/health", "/v3/api-docs/**", "/swagger-ui/**", "/swagger-ui.html").permitAll()
                .anyRequest().authenticated())
            .exceptionHandling(e -> e
                .authenticationEntryPoint((req, res, ex) -> write(om, res, 401, new ApiError("UNAUTHENTICATED", "Cần đăng nhập", null)))
                .accessDeniedHandler((req, res, ex) -> write(om, res, 403, new ApiError("FORBIDDEN", "Không đủ quyền", null))))
            .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }

    private static void write(ObjectMapper om, jakarta.servlet.http.HttpServletResponse res, int status, ApiError body)
            throws java.io.IOException {
        res.setStatus(status);
        res.setContentType(MediaType.APPLICATION_JSON_VALUE + ";charset=UTF-8");
        om.writeValue(res.getOutputStream(), body);
    }

    @Bean
    RoleHierarchy roleHierarchy() {
        return RoleHierarchyImpl.fromHierarchy("ROLE_ADMIN > ROLE_EDITOR\nROLE_EDITOR > ROLE_VIEWER");
    }

    @Bean
    static MethodSecurityExpressionHandler methodSecurityExpressionHandler(RoleHierarchy rh) {
        DefaultMethodSecurityExpressionHandler h = new DefaultMethodSecurityExpressionHandler();
        h.setRoleHierarchy(rh);
        return h;
    }

    @Bean
    PasswordEncoder passwordEncoder() { return new BCryptPasswordEncoder(); }
}
```

```java
package vn.giapha.auth;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.util.Base64;
import java.util.HexFormat;

final class TokenHasher {
    private static final SecureRandom RANDOM = new SecureRandom();

    private TokenHasher() {}

    static String newToken() {
        byte[] b = new byte[32];
        RANDOM.nextBytes(b);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(b);
    }

    static String hash(String token) {
        try {
            return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(token.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }
}
```

```java
package vn.giapha.auth;

import java.time.Duration;
import java.time.Instant;
import java.util.Locale;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.giapha.common.ApiException;

@Service
public class AuthService {
    static final int MAX_FAILED = 5;
    static final Duration LOCK_FOR = Duration.ofMinutes(15);

    public record UserView(long id, String username, String displayName, Role role, Long linkedPersonId) {
        static UserView of(AppUser u) { return new UserView(u.getId(), u.getUsername(), u.getDisplayName(), u.getRole(), u.getLinkedPersonId()); }
    }
    public record LoginResult(String accessToken, long expiresInSeconds, String refreshToken, UserView user) {}

    private final AppUserRepository users;
    private final RefreshTokenRepository tokens;
    private final PasswordEncoder encoder;
    private final JwtService jwt;
    private final long refreshDays;

    public AuthService(AppUserRepository users, RefreshTokenRepository tokens, PasswordEncoder encoder, JwtService jwt,
                       @Value("${app.jwt.refresh-days}") long refreshDays) {
        this.users = users;
        this.tokens = tokens;
        this.encoder = encoder;
        this.jwt = jwt;
        this.refreshDays = refreshDays;
    }

    @Transactional
    public AppUser register(String username, String password, String displayName) {
        String name = username.toLowerCase(Locale.ROOT);
        if (users.existsByUsername(name)) throw new ApiException(HttpStatus.CONFLICT, "USERNAME_TAKEN", "Tên đăng nhập đã tồn tại");
        AppUser u = new AppUser();
        u.setUsername(name);
        u.setPasswordHash(encoder.encode(password));
        u.setDisplayName(displayName);
        u.setRole(Role.VIEWER);
        u.setStatus(UserStatus.PENDING);
        return users.save(u);
    }

    /** noRollbackFor: bộ đếm sai mật khẩu phải được lưu dù request trả lỗi. */
    @Transactional(noRollbackFor = ApiException.class)
    public LoginResult login(String username, String password) {
        AppUser u = users.findByUsername(username.toLowerCase(Locale.ROOT))
            .orElseThrow(() -> new ApiException(HttpStatus.UNAUTHORIZED, "BAD_CREDENTIALS", "Sai tên đăng nhập hoặc mật khẩu"));
        Instant now = Instant.now();
        if (u.getLockedUntil() != null && u.getLockedUntil().isAfter(now)) {
            throw new ApiException(HttpStatus.valueOf(423), "ACCOUNT_LOCKED", "Tài khoản tạm khoá, thử lại sau");
        }
        if (u.getStatus() == UserStatus.LOCKED) {
            throw new ApiException(HttpStatus.valueOf(423), "ACCOUNT_LOCKED", "Tài khoản đã bị khoá");
        }
        if (!encoder.matches(password, u.getPasswordHash())) {
            u.setFailedAttempts(u.getFailedAttempts() + 1);
            if (u.getFailedAttempts() >= MAX_FAILED) {
                u.setLockedUntil(now.plus(LOCK_FOR));
                u.setFailedAttempts(0);
            }
            users.save(u);
            throw new ApiException(HttpStatus.UNAUTHORIZED, "BAD_CREDENTIALS", "Sai tên đăng nhập hoặc mật khẩu");
        }
        if (u.getStatus() == UserStatus.PENDING) {
            throw new ApiException(HttpStatus.FORBIDDEN, "ACCOUNT_PENDING", "Tài khoản đang chờ duyệt");
        }
        u.setFailedAttempts(0);
        u.setLockedUntil(null);
        users.save(u);
        return issue(u);
    }

    @Transactional
    public LoginResult refresh(String rawRefresh) {
        RefreshToken t = find(rawRefresh);
        if (t == null || t.isRevoked() || t.getExpiresAt().isBefore(Instant.now())) throw unauth();
        AppUser u = users.findById(t.getUserId()).orElseThrow(this::unauth);
        if (u.getStatus() != UserStatus.ACTIVE) throw unauth();
        t.setRevoked(true);
        tokens.save(t);
        return issue(u);
    }

    @Transactional
    public void logout(String rawRefresh) {
        RefreshToken t = find(rawRefresh);
        if (t != null) { t.setRevoked(true); tokens.save(t); }
    }

    private RefreshToken find(String raw) {
        if (raw == null || raw.isBlank()) return null;
        return tokens.findByTokenHash(TokenHasher.hash(raw)).orElse(null);
    }

    private LoginResult issue(AppUser u) {
        String raw = TokenHasher.newToken();
        RefreshToken t = new RefreshToken();
        t.setUserId(u.getId());
        t.setTokenHash(TokenHasher.hash(raw));
        t.setExpiresAt(Instant.now().plus(Duration.ofDays(refreshDays)));
        tokens.save(t);
        return new LoginResult(jwt.issue(u.getId(), u.getRole()), jwt.accessSeconds(), raw, UserView.of(u));
    }

    private ApiException unauth() { return new ApiException(HttpStatus.UNAUTHORIZED, "UNAUTHENTICATED", "Phiên đăng nhập hết hạn"); }
}
```

```java
import org.springframework.web.bind.annotation.*;

public class AuthController {
    static final String COOKIE = "refresh_token";

    public record RegisterRequest(
        @NotBlank @Size(min = 3, max = 50) @Pattern(regexp = "[a-zA-Z0-9._-]+") String username,
        @NotBlank @Size(min = 8, max = 72) String password,
        @Size(max = 120) String displayName) {}

    public record RegisterResponse(long id, UserStatus status) {}
    public record TokenResponse(String accessToken, long expiresInSeconds, AuthService.UserView user) {}

    private final AuthService auth;
    private final boolean secure;
    private final long refreshDays;

    public AuthController(AuthService auth, @Value("${app.cookie-secure}") boolean secure,
                          @Value("${app.jwt.refresh-days}") long refreshDays) {
        this.auth = auth;
        this.secure = secure;
        this.refreshDays = refreshDays;
    }

    @PostMapping("/register")
    @ResponseStatus(HttpStatus.CREATED)
    public RegisterResponse register(@Valid @RequestBody RegisterRequest r) {
        AppUser u = auth.register(r.username(), r.password(), r.displayName());
        return new RegisterResponse(u.getId(), u.getStatus());
    }

    @PostMapping("/login")
    public ResponseEntity<TokenResponse> login(@Valid @RequestBody LoginRequest r) {
        return withCookie(auth.login(r.username(), r.password()));
    }

    @PostMapping("/refresh")
    public ResponseEntity<TokenResponse> refresh(@CookieValue(name = COOKIE, required = false) String raw) {
        return withCookie(auth.refresh(raw));
    }

    @PostMapping("/logout")
    public ResponseEntity<Void> logout(@CookieValue(name = COOKIE, required = false) String raw) {
        auth.logout(raw);
        return ResponseEntity.noContent().header(HttpHeaders.SET_COOKIE, cookie("", Duration.ZERO).toString()).build();
    }

    private ResponseEntity<TokenResponse> withCookie(AuthService.LoginResult r) {
        return ResponseEntity.ok()
            .header(HttpHeaders.SET_COOKIE, cookie(r.refreshToken(), Duration.ofDays(refreshDays)).toString())
            .body(new TokenResponse(r.accessToken(), r.expiresInSeconds(), r.user()));
    }

    private ResponseCookie cookie(String value, Duration maxAge) {
        return ResponseCookie.from(COOKIE, value).httpOnly(true).secure(secure).sameSite("Strict")
            .path("/api/v1/auth").maxAge(maxAge).build();
    }
```

```ts
export type ApiErrorBody = { code: string; message: string; details: unknown }
export type ApiFailure = ApiErrorBody & { status: number }

let accessToken: string | null = null
let unauthorizedHandler: (() => Promise<boolean>) | null = null
export const setAccessToken = (t: string | null) => { accessToken = t }
export const getAccessToken = () => accessToken
export const setUnauthorizedHandler = (h: (() => Promise<boolean>) | null) => { unauthorizedHandler = h }

export async function http<T>(path: string, init: RequestInit = {}, retried = false): Promise<T> {
  const headers = new Headers(init.headers)
  if (init.body && !(init.body instanceof FormData) && !headers.has('Content-Type')) {
    headers.set('Content-Type', 'application/json')
  }
  if (accessToken) headers.set('Authorization', `Bearer ${accessToken}`)
  const res = await fetch(`/api/v1${path}`, { ...init, headers, credentials: 'include' })
  if (res.status === 401 && !retried && unauthorizedHandler && !path.startsWith('/auth/')) {
    if (await unauthorizedHandler()) return http<T>(path, init, true)
  }
  if (!res.ok) {
    let body: ApiErrorBody = { code: 'UNKNOWN', message: res.statusText, details: null }
    try { body = await res.json() } catch { /* giữ mặc định */ }
    throw { ...body, status: res.status } satisfies ApiFailure
  }
  if (res.status === 204) return undefined as T
  return (await res.json()) as T
}
```

```tsx
import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from 'react'
import { authApi, type CurrentUser, type TokenResponse } from '../api/auth'
import { setAccessToken, setUnauthorizedHandler } from '../api/http'

export interface AuthState {
  user: CurrentUser | null
  loading: boolean
  login: (username: string, password: string) => Promise<void>
  logout: () => Promise<void>
}

export const AuthContext = createContext<AuthState>({ user: null, loading: true, login: async () => {}, logout: async () => {} })
export const useAuth = () => useContext(AuthContext)

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<CurrentUser | null>(null)
  const [loading, setLoading] = useState(true)

  const accept = useCallback((r: TokenResponse) => { setAccessToken(r.accessToken); setUser(r.user) }, [])
  const clear = useCallback(() => { setAccessToken(null); setUser(null) }, [])

  useEffect(() => {
    setUnauthorizedHandler(async () => {
      try { accept(await authApi.refresh()); return true } catch { clear(); return false }
    })
    authApi.refresh().then(accept).catch(clear).finally(() => setLoading(false))
    return () => setUnauthorizedHandler(null)
  }, [accept, clear])

  const value = useMemo<AuthState>(() => ({
    user, loading,
    login: async (u, p) => accept(await authApi.login(u, p)),
    logout: async () => { try { await authApi.logout() } finally { clear() } },
  }), [user, loading, accept, clear])

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}
```

```tsx
    <div style={{ padding: 16 }}>
      <Space style={{ marginBottom: 12 }}>
        <span>Lọc theo trạng thái:</span>
        <Select<UserStatus | 'ALL'> style={{ width: 160 }} value={filter ?? 'ALL'}
          onChange={(v) => setFilter(v === 'ALL' ? undefined : v)}
          options={[{ value: 'ALL', label: 'Tất cả' }, ...(['PENDING', 'ACTIVE', 'LOCKED'] as UserStatus[]).map((s) => ({ value: s, label: STATUS_TEXT[s] }))]} />
      </Space>
      <Table<AdminUser> rowKey="id" loading={isLoading} dataSource={data} size="small" scroll={{ x: true }} pagination={{ pageSize: 20 }}
        columns={[
          { title: 'Tên đăng nhập', dataIndex: 'username' },
          { title: 'Họ tên', dataIndex: 'displayName' },
          { title: 'Trạng thái', dataIndex: 'status', render: (s: UserStatus) => <Tag color={STATUS_COLOR[s]}>{STATUS_TEXT[s]}</Tag> },
          { title: 'Vai trò', dataIndex: 'role', render: (r: Role, row) => (
            <Select<Role> size="small" value={r} style={{ width: 110 }} onChange={(role) => mut.mutate({ id: row.id, patch: { role } })}
              options={[{ value: 'VIEWER', label: 'Cấp thấp' }, { value: 'EDITOR', label: 'Cấp trung' }, { value: 'ADMIN', label: 'Cấp cao' }]} />) },
          { title: 'Gắn với người (ID)', dataIndex: 'linkedPersonId', render: (v: number | null, row) => (
            <InputNumber size="small" min={1} defaultValue={v ?? undefined} style={{ width: 100 }}
              onBlur={(e) => { const n = Number(e.target.value); if (n && n !== v) mut.mutate({ id: row.id, patch: { linkedPersonId: n } }) }} />) },
          { title: 'Thao tác', render: (_, row) => (
            <Space>
              {row.status !== 'ACTIVE' && <Button size="small" type="primary" onClick={() => mut.mutate({ id: row.id, patch: { status: 'ACTIVE' } })}>{row.status === 'PENDING' ? 'Duyệt' : 'Mở khoá'}</Button>}
              {row.status === 'ACTIVE' && <Button size="small" danger onClick={() => mut.mutate({ id: row.id, patch: { status: 'LOCKED' } })}>Khoá</Button>}
            </Space>) },
        ]} />
    </div>
  )
}
```
