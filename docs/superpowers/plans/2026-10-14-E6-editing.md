## Global Constraints

- Ma trận quyền (nguồn chuẩn): thêm người = EDITOR+; sửa hồ sơ/ảnh/tiểu sử = ADMIN, hoặc EDITOR với người **mình tạo** và `now - created_at ≤ 24h`; thêm/gỡ quan hệ, hôn phối, đặt thứ tự thủ công = EDITOR+ (mọi người); xoá mềm, quản lý dòng họ = ADMIN. VIEWER không ghi được gì.
- Mã lỗi: `FORBIDDEN`(403), `EDIT_WINDOW_EXPIRED`(403), `VERSION_CONFLICT`(409), `RELATION_CYCLE`(409), `RELATION_INVALID`(409), `VALIDATION_FAILED`(400).
- Sửa người bắt buộc gửi `version`; sai lệch → `VERSION_CONFLICT`, không ghi gì.
- Ảnh: chỉ `image/jpeg` hoặc `image/png`, ≤ 10 MB, phải giải mã được; sinh thumbnail vuông cắt giữa 96 px và 256 px (JPEG); khoá lưu `p/{personId}/{photoKey}/{orig|96|256}`; `photo_key` là UUID mới mỗi lần đổi ảnh.
- Mọi thao tác ghi `audit_log` với `entity_type ∈ {PERSON, PARENT_CHILD, UNION}`, `action ∈ {CREATE, UPDATE, DELETE, SET_BIRTH_ORDER, SET_PHOTO}`, `before_data`/`after_data` là snapshot JSON.
- Sau thay đổi quan hệ phải dựng lại `clan_member` của mọi dòng họ chứa các người liên quan (`ClanMemberService.rebuildClansOf`).
- Không xoá mềm thủy tổ của một dòng họ (`RELATION_INVALID`).
- Ngày không đầy đủ: `MONTH` lưu ngày 01, `YEAR` lưu 01-01.

package vn.giapha.audit;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;

@Service
public class AuditService {
    private final JdbcTemplate jdbc;
    private final ObjectMapper om;

    public AuditService(JdbcTemplate jdbc, ObjectMapper om) {
        this.jdbc = jdbc;
        this.om = om;
    }

    public void record(Long userId, String entityType, long entityId, String action, Object before, Object after) {
        jdbc.update("INSERT INTO audit_log(user_id, entity_type, entity_id, action, before_data, after_data) VALUES (?,?,?,?,?::jsonb,?::jsonb)",
            userId, entityType, entityId, action, json(before), json(after));
    }

    private String json(Object o) {
        if (o == null) return null;
        try { return om.writeValueAsString(o); }
        catch (JsonProcessingException e) { throw new IllegalStateException(e); }
    }
}

package vn.giapha.person;

import java.time.Clock;
import java.time.Duration;
import java.time.Instant;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Component;
import vn.giapha.auth.AuthUser;
import vn.giapha.auth.Role;
import vn.giapha.common.ApiException;

@Component
public class EditPolicy {
    static final Duration WINDOW = Duration.ofHours(24);
    private final Clock clock;

    public EditPolicy(Clock clock) { this.clock = clock; }

    public boolean canEdit(AuthUser me, Person p) {
        if (me.role() == Role.ADMIN) return true;
        if (me.role() != Role.EDITOR) return false;
        return isOwner(me, p) && inWindow(p);
    }

    public void assertCanEdit(AuthUser me, Person p) {
        if (me.role() == Role.ADMIN) return;
        if (me.role() != Role.EDITOR || !isOwner(me, p)) {
            throw new ApiException(HttpStatus.FORBIDDEN, "FORBIDDEN", "Chỉ được sửa người do mình tạo");
        }
        if (!inWindow(p)) {
            throw new ApiException(HttpStatus.FORBIDDEN, "EDIT_WINDOW_EXPIRED", "Đã quá 24 giờ, chỉ quản trị viên được sửa");
        }
    }

    private boolean isOwner(AuthUser me, Person p) { return p.getCreatedBy() != null && p.getCreatedBy() == me.id(); }

    private boolean inWindow(Person p) {
        Instant created = p.getCreatedAt();
        return created != null && !Instant.now(clock).isAfter(created.plus(WINDOW));
    }
}

```java
package vn.giapha.person;

import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.giapha.audit.AuditService;
import vn.giapha.audit.Snapshots;
import vn.giapha.auth.AuthUser;
import vn.giapha.clan.ClanMemberService;
import vn.giapha.common.ApiException;

@Service
public class PersonService {
    private final PersonRepository persons;
    private final EditPolicy policy;
    private final AuditService audit;
    private final ClanMemberService clanMembers;
    private final JdbcTemplate jdbc;

    public PersonService(PersonRepository persons, EditPolicy policy, AuditService audit, ClanMemberService clanMembers, JdbcTemplate jdbc) {
        this.persons = persons;
        this.policy = policy;
        this.audit = audit;
        this.clanMembers = clanMembers;
        this.jdbc = jdbc;
    }

    @Transactional
    public Person create(AuthUser me, PersonInput in) {
        Person p = new Person();
        apply(p, in);
        p.setCreatedBy(me.id());
        p = persons.saveAndFlush(p);
        audit.record(me.id(), "PERSON", p.getId(), "CREATE", null, Snapshots.of(p));
        return p;
    }

    @Transactional
    public Person update(AuthUser me, long id, PersonInput in) {
        Person p = persons.findByIdAndDeletedAtIsNull(id).orElseThrow(this::notFound);
        policy.assertCanEdit(me, p);
        if (in.version() == null) throw bad("Thiếu version");
        if (!in.version().equals(p.getVersion())) {
            throw new ApiException(HttpStatus.CONFLICT, "VERSION_CONFLICT", "Dữ liệu đã được người khác sửa, vui lòng tải lại");
        }
        var before = Snapshots.of(p);
        apply(p, in);
        p = persons.saveAndFlush(p);
        audit.record(me.id(), "PERSON", id, "UPDATE", before, Snapshots.of(p));
        return p;
    }

    @Transactional
    public void delete(AuthUser me, long id) {
        Person p = persons.findByIdAndDeletedAtIsNull(id).orElseThrow(this::notFound);
        Integer founders = jdbc.queryForObject("SELECT count(*) FROM clan WHERE founder_person_id = ?", Integer.class, id);
        if (founders != null && founders > 0) {
            throw new ApiException(HttpStatus.CONFLICT, "RELATION_INVALID", "Không thể xoá thủy tổ của một dòng họ");
        }
        List<Long> affected = List.of(id);
        var before = Snapshots.of(p);
        p.setDeletedAt(java.time.Instant.now());
        persons.saveAndFlush(p);
        jdbc.update("UPDATE app_user SET linked_person_id = NULL WHERE linked_person_id = ?", id);
        clanMembers.rebuildClansOf(affected);   // chạy khi người còn trong clan_member để biết dòng họ nào cần dựng lại
        audit.record(me.id(), "PERSON", id, "DELETE", before, Snapshots.of(p));
    }

    private void apply(Person p, PersonInput in) {
        boolean alive = in.alive() == null || in.alive();
        if (alive && in.deathDate() != null) throw bad("Người còn sống không có ngày mất");
        if (in.birthDate() != null && in.deathDate() != null && in.deathDate().isBefore(in.birthDate())) throw bad("Ngày mất trước ngày sinh");
        p.setFullName(in.fullName().trim());
        p.setNickname(blank(in.nickname()));
        p.setGender(in.gender());
        p.setBirthDate(in.birthDate());
        p.setBirthPrecision(in.birthDate() == null ? DatePrecision.UNKNOWN : in.birthPrecision() == null ? DatePrecision.DAY : in.birthPrecision());
        p.setBirthLunar(blank(in.birthLunar()));
        p.setAlive(alive);
        p.setDeathDate(alive ? null : in.deathDate());
        p.setDeathPrecision(alive || in.deathDate() == null ? DatePrecision.UNKNOWN : in.deathPrecision() == null ? DatePrecision.DAY : in.deathPrecision());
        p.setDeathAnniversaryLunar(blank(in.deathAnniversaryLunar()));
        p.setOriginPlace(blank(in.originPlace()));
        p.setHomePlace(blank(in.homePlace()));
        p.setBurialPlace(blank(in.burialPlace()));
        p.setBiography(in.biography() == null || in.biography().isBlank() ? null : in.biography());
    }

    private static String blank(String s) { return s == null || s.isBlank() ? null : s.trim(); }
    private ApiException bad(String m) { return new ApiException(HttpStatus.BAD_REQUEST, "VALIDATION_FAILED", m); }
    private ApiException notFound() { return new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND", "Không tìm thấy người này"); }
}
```
Quan trọng: trong `delete`, `rebuildClansOf` phải gọi **sau** khi đánh dấu `deleted_at` (để `rebuild` loại người đó) nhưng cần biết dòng họ chứa họ **trước** khi `clan_member` bị xoá bởi `rebuild`. Vì `rebuildClansOf` truy vấn `clan_member` trước rồi mới dựng lại, thứ tự trong code trên là đúng (đã `saveAndFlush` deleted_at trước, `clan_member` vẫn còn dòng cũ khi truy vấn).

import java.time.LocalDate;

import vn.giapha.auth.AuthUser;
import vn.giapha.clan.ClanMemberService;
import vn.giapha.common.ApiException;

@Service
public class RelationService {
    private final ParentChildService pcs;
    private final UnionService unions;
    private final ParentChildRepository links;
    private final UnionRepository unionRepo;
    private final ClanMemberService clans;
    private final AuditService audit;

    public RelationService(ParentChildService pcs, UnionService unions, ParentChildRepository links, UnionRepository unionRepo,
                           ClanMemberService clans, AuditService audit) {
        this.pcs = pcs;
        this.unions = unions;
        this.links = links;
        this.unionRepo = unionRepo;
        this.clans = clans;
        this.audit = audit;
    }

    @Transactional
    public ParentChildResult addParentChild(AuthUser me, long parentId, long childId, ParentKind kind, Long unionId) {
        ParentChildResult r = pcs.add(parentId, childId, kind, unionId);
        audit.record(me.id(), "PARENT_CHILD", r.link().getId(), "CREATE", null, Snapshots.of(r.link()));
        clans.rebuildClansOf(List.of(parentId, childId));
        return r;
    }

    @Transactional
    public List<ParentChildResult> addChild(AuthUser me, long parentId, long childId, ParentKind kind, Long unionId) {
        List<ParentChildResult> out = new ArrayList<>();
        out.add(pcs.add(parentId, childId, kind, unionId));
        if (unionId != null) {
            UnionRel u = unionRepo.findById(unionId).orElseThrow(() -> new ApiException(HttpStatus.CONFLICT, "RELATION_INVALID", "Không tìm thấy hôn phối"));
            long other = u.getPartnerAId() == parentId ? u.getPartnerBId() : u.getPartnerAId();
            out.add(pcs.add(other, childId, kind, unionId));
        }
        out.forEach(r -> audit.record(me.id(), "PARENT_CHILD", r.link().getId(), "CREATE", null, Snapshots.of(r.link())));
        List<Long> involved = new ArrayList<>(List.of(parentId, childId));
        out.forEach(r -> involved.add(r.link().getParentId()));
        clans.rebuildClansOf(involved);
        return out;
    }

    @Transactional
    public void removeParentChild(AuthUser me, long linkId) {
        ParentChild l = links.findById(linkId).orElseThrow(this::notFound);
        var before = Snapshots.of(l);
        pcs.remove(linkId);
        audit.record(me.id(), "PARENT_CHILD", linkId, "DELETE", before, null);
        clans.rebuildClansOf(List.of(l.getParentId(), l.getChildId()));
    }

    @Transactional
    public UnionRel createUnion(AuthUser me, long a, long b, Integer order, UnionStatus status, LocalDate start, LocalDate end) {
        UnionRel u = unions.create(a, b, order, status, start, end);
        audit.record(me.id(), "UNION", u.getId(), "CREATE", null, Snapshots.of(u));
        clans.rebuildClansOf(List.of(a, b));
        return u;
    }

    @Transactional
    public UnionRel updateUnion(AuthUser me, long id, Integer order, UnionStatus status, LocalDate start, LocalDate end) {
        UnionRel existing = unionRepo.findById(id).orElseThrow(this::notFound);
        var before = Snapshots.of(existing);
        UnionRel u = unions.update(id, order, status, start, end);
        audit.record(me.id(), "UNION", id, "UPDATE", before, Snapshots.of(u));
        return u;
    }

    @Transactional
    public void deleteUnion(AuthUser me, long id) {
        UnionRel u = unionRepo.findById(id).orElseThrow(this::notFound);
        var before = Snapshots.of(u);
        unions.delete(id);
        audit.record(me.id(), "UNION", id, "DELETE", before, null);
        clans.rebuildClansOf(List.of(u.getPartnerAId(), u.getPartnerBId()));
    }

    private ApiException notFound() { return new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND", "Không tìm thấy"); }
}

public class ThumbnailService {
    public byte[] square(byte[] image, int size) {
        BufferedImage src;
        try { src = ImageIO.read(new ByteArrayInputStream(image)); }
        catch (IOException e) { src = null; }
        if (src == null) throw new ApiException(HttpStatus.BAD_REQUEST, "VALIDATION_FAILED", "Tệp không phải ảnh hợp lệ");
        int side = Math.min(src.getWidth(), src.getHeight());
        BufferedImage crop = src.getSubimage((src.getWidth() - side) / 2, (src.getHeight() - side) / 2, side, side);
        BufferedImage out = new BufferedImage(size, size, BufferedImage.TYPE_INT_RGB);
        var g = out.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
        g.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
        g.setColor(Color.WHITE);
        g.fillRect(0, 0, size, size);
        g.drawImage(crop, 0, 0, size, size, null);
        g.dispose();
        try {
            ByteArrayOutputStream bos = new ByteArrayOutputStream();
            ImageIO.write(out, "jpg", bos);
            return bos.toByteArray();
        } catch (IOException e) {
            throw new IllegalStateException(e);
        }

package vn.giapha.storage;

import java.net.URI;
import java.util.Optional;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import software.amazon.awssdk.auth.credentials.AwsBasicCredentials;
import software.amazon.awssdk.auth.credentials.StaticCredentialsProvider;
import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.S3Configuration;
import software.amazon.awssdk.services.s3.model.*;

@Component
public class S3PhotoStorage implements PhotoStorage {
    private final S3Client s3;
    private final String bucket;
    private volatile boolean bucketReady;

    public S3PhotoStorage(@Value("${app.storage.endpoint}") String endpoint, @Value("${app.storage.access-key}") String accessKey,
                          @Value("${app.storage.secret-key}") String secretKey, @Value("${app.storage.bucket}") String bucket) {
        this.bucket = bucket;
        this.s3 = S3Client.builder().endpointOverride(URI.create(endpoint)).region(Region.US_EAST_1)
            .credentialsProvider(StaticCredentialsProvider.create(AwsBasicCredentials.create(accessKey, secretKey)))
            .serviceConfiguration(S3Configuration.builder().pathStyleAccessEnabled(true).build()).build();
    }

    private void ensureBucket() {
        if (bucketReady) return;
        try { s3.headBucket(HeadBucketRequest.builder().bucket(bucket).build()); }
        catch (NoSuchBucketException e) { s3.createBucket(CreateBucketRequest.builder().bucket(bucket).build()); }
        bucketReady = true;
    }

        storage.put(key(personId, newKey, "orig"), data, file.getContentType());

import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.giapha.common.ApiException;
import vn.giapha.person.PersonRepository;

@Service
public class ClanAdminService {
    public record ClanView(long id, String name, Long founderPersonId, String description,
                           SiblingCountMode siblingCountMode, boolean useUtLabel, int memberCount) {}

    private final ClanRepository clans;
    private final ClanMemberRepository members;
    private final ClanMemberService memberService;
    private final PersonRepository persons;

    public ClanAdminService(ClanRepository clans, ClanMemberRepository members, ClanMemberService memberService, PersonRepository persons) {
        this.clans = clans;
        this.members = members;
        this.memberService = memberService;
        this.persons = persons;
    }

    public List<ClanView> list() { return clans.findAll().stream().map(this::view).toList(); }

    @Transactional
    public ClanView create(String name, Long founderId, String description, SiblingCountMode mode, Boolean useUt) {
        Clan c = new Clan();
        apply(c, name, founderId, description, mode, useUt);
        c = clans.saveAndFlush(c);
        if (c.getFounderPersonId() != null) memberService.rebuild(c.getId());
        return view(c);
    }

    @Transactional
    public ClanView update(long id, String name, Long founderId, String description, SiblingCountMode mode, Boolean useUt) {
        Clan c = clans.findById(id).orElseThrow(() -> new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND", "Không tìm thấy dòng họ"));
        apply(c, name, founderId, description, mode, useUt);
        c = clans.saveAndFlush(c);
        if (c.getFounderPersonId() != null) memberService.rebuild(c.getId());
        return view(c);
    }

    private void apply(Clan c, String name, Long founderId, String description, SiblingCountMode mode, Boolean useUt) {
        if (founderId != null && persons.findByIdAndDeletedAtIsNull(founderId).isEmpty()) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "VALIDATION_FAILED", "Thủy tổ không tồn tại");
        }
        c.setName(name.trim());
        c.setFounderPersonId(founderId);
        c.setDescription(description);
        if (mode != null) c.setSiblingCountMode(mode);
        if (useUt != null) c.setUseUtLabel(useUt);
    }

    private ClanView view(Clan c) {
        return new ClanView(c.getId(), c.getName(), c.getFounderPersonId(), c.getDescription(), c.getSiblingCountMode(),
            c.isUseUtLabel(), members.findByClanId(c.getId()).size());
    }
}

import dayjs, { type Dayjs } from 'dayjs'

export type Precision = 'DAY' | 'MONTH' | 'YEAR' | 'UNKNOWN'

export function toIso(d: Dayjs | null | undefined, precision: Precision): string | null {
  if (!d || precision === 'UNKNOWN') return null
  if (precision === 'YEAR') return `${d.format('YYYY')}-01-01`
  if (precision === 'MONTH') return `${d.format('YYYY-MM')}-01`
  return d.format('YYYY-MM-DD')
}

export function fromIso(s: string | null | undefined): Dayjs | null {
  return s ? dayjs(s) : null

    const { selectionStart: s, selectionEnd: e } = el

  }

  return (

      <Space style={{ marginBottom: 4 }}>
        <Button size="small" onClick={() => wrap('**')}><b>B</b></Button>
        <Button size="small" onClick={() => wrap('*')}><i>I</i></Button>
        <Button size="small" onClick={bullet}>• Danh sách</Button>
      </Space>
      <Input.TextArea ref={ref as never} value={value} onChange={(e) => onChange?.(e.target.value)} rows={8} maxLength={20000} showCount />

  )

import { useQuery, useQueryClient } from '@tanstack/react-query'
import { Alert, App, Modal, Spin, Tabs } from 'antd'
import { personApi } from '../../api/person'
import { editApi } from '../../api/edit'

export default function PersonDialog({ personId, onClose, onNavigate, onTreeChanged }: {
  personId: number | null; onClose(): void; onNavigate(id: number): void; onTreeChanged(): void
}) {
  const { user } = useAuth()
  const { message } = App.useApp()
  const qc = useQueryClient()
  const [editing, setEditing] = useState(false)
