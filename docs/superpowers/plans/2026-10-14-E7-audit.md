**Goal:** Cấp trung và cấp cao xem được "Lịch sử sửa" của một người (ai sửa gì, lúc nào, thay đổi cụ thể); cấp cao khôi phục được bản cũ khi sửa nhầm (hồ sơ, thứ tự thủ công, xoá/khôi phục người, thêm/gỡ quan hệ và hôn phối). Mọi lần khôi phục cũng được ghi nhật ký.

**Architecture:** `audit_log` đã được ghi ở E6. E7 thêm: `AuditSummary` (hàm thuần tạo mô tả tiếng Việt + danh sách thay đổi từ snapshot JSON), `AuditQueryService` (truy vấn theo người, phân trang bằng `beforeId`), `RevertService` (áp lại snapshot `before` theo từng loại thực thể, dùng lại các service E2/E6 để vẫn kiểm vòng lặp và ràng buộc), `AuditController`. **Spec:** `docs/superpowers/specs/2026-10-14-gia-pha-design.md`; tài liệu gốc mục "Phân quyền" (Xem lịch sử sửa: ADMIN, EDITOR; Khôi phục bản cũ: ADMIN), "Mô hình dữ liệu" (`audit_log`).

## Global Constraints

- `GET /api/v1/audit?entityId=` : EDITOR+; không có `entityId` thì chỉ ADMIN (nguồn cấp toàn hệ thống). `POST /api/v1/audit/{id}/revert` : ADMIN.
- Mã lỗi mới `[BỔ SUNG]`: `NOT_REVERTIBLE` (409) — mục không khôi phục được (ảnh, mục `REVERT`, quan hệ đã bị gỡ…).
- Phân trang: `limit` mặc định 30, kẹp `[1,100]`; `beforeId` lấy các mục có `id < beforeId`; sắp xếp `id` giảm dần.
- Khôi phục **không** ghi đè ảnh (`photoKey`) và `deleted`; khôi phục `SET_PHOTO` không hỗ trợ (ảnh cũ đã bị xoá khỏi kho).
- Mỗi lần khôi phục ghi một dòng `audit_log` với `action = 'REVERT'`.
- Lịch sử của một người gồm: bản ghi `PERSON` của họ + bản ghi `PARENT_CHILD`/`UNION` có họ là một đầu của quan hệ.
- VIEWER không thấy tab Lịch sử sửa và API trả 403.

## Review Focus

- Khôi phục một lần sửa cũ khi đã có nhiều lần sửa sau đó: trả về đúng giá trị `before` của lần sửa được chọn, không làm mất `photoKey`/`version` hợp lệ.
- Khôi phục "gỡ quan hệ" mà hiện tại sẽ tạo vòng lặp: `RELATION_CYCLE`, không thay đổi gì, không ghi `REVERT`.
- Khôi phục "thêm quan hệ" khi quan hệ đã bị gỡ tay: `NOT_REVERTIBLE`, không lỗi 500.
- Khôi phục xoá người → người xuất hiện lại trong cây dòng họ (clan_member dựng lại), kể cả con cháu đã bị cắt theo.
- Khôi phục mục `REVERT` hoặc `SET_PHOTO` → `NOT_REVERTIBLE`.
- Người xem lịch sử là EDITOR không thấy nút Khôi phục (chỉ ADMIN).
- Biography dài không làm bảng thay đổi vỡ giao diện (cắt ngắn trong phần diff).

```java
package vn.giapha.audit;

import com.fasterxml.jackson.databind.JsonNode;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.function.Function;

public final class AuditSummary {
    public record Change(String field, String label, String from, String to) {}

    private static final Map<String, String> PERSON_FIELDS = new LinkedHashMap<>();
    private static final Map<String, String> UNION_FIELDS = new LinkedHashMap<>();
    static {
        PERSON_FIELDS.put("fullName", "Họ tên");
        PERSON_FIELDS.put("nickname", "Tên thường gọi");
        PERSON_FIELDS.put("gender", "Giới tính");
        PERSON_FIELDS.put("birthDate", "Ngày sinh");
        PERSON_FIELDS.put("birthPrecision", "Độ chính xác ngày sinh");
        PERSON_FIELDS.put("birthLunar", "Ngày sinh (âm lịch)");
        PERSON_FIELDS.put("alive", "Còn sống");
        PERSON_FIELDS.put("deathDate", "Ngày mất");
        PERSON_FIELDS.put("deathPrecision", "Độ chính xác ngày mất");
        PERSON_FIELDS.put("deathAnniversaryLunar", "Ngày giỗ (âm lịch)");
        PERSON_FIELDS.put("originPlace", "Nguyên quán");
        PERSON_FIELDS.put("homePlace", "Quê quán");
        PERSON_FIELDS.put("burialPlace", "Nơi an táng");
        PERSON_FIELDS.put("biography", "Sơ yếu lý lịch");
        PERSON_FIELDS.put("birthOrderOverride", "Thứ tự thủ công");
        UNION_FIELDS.put("order", "Thứ tự hôn phối");
        UNION_FIELDS.put("status", "Tình trạng hôn phối");
        UNION_FIELDS.put("startDate", "Ngày bắt đầu");
        UNION_FIELDS.put("endDate", "Ngày kết thúc");
    }

    private AuditSummary() {}

    public static List<Change> changes(String entityType, JsonNode before, JsonNode after) {
        Map<String, String> fields = "PERSON".equals(entityType) ? PERSON_FIELDS : "UNION".equals(entityType) ? UNION_FIELDS : Map.of();
        List<Change> out = new ArrayList<>();
        if (before == null || after == null || fields.isEmpty()) return out;
        fields.forEach((f, label) -> {
            String a = show(f, before.get(f)), b = show(f, after.get(f));
            if (!a.equals(b)) out.add(new Change(f, label, a, b));
        });
        return out;
    }

    public static String summary(String entityType, String action, JsonNode before, JsonNode after, Function<Long, String> nameOf) {
        JsonNode any = after != null ? after : before;
        switch (entityType) {
            case "PERSON": {
                String name = text(after != null && after.hasNonNull("fullName") ? after : before, "fullName");
                switch (action) {
                    case "CREATE": return "Tạo người «" + name + "»";
                    case "UPDATE": return "Sửa hồ sơ «" + name + "»";
                    case "DELETE": return "Xoá «" + name + "»";
                    case "SET_PHOTO": return "Đổi ảnh «" + name + "»";
                    case "REVERT": return "Khôi phục bản cũ «" + name + "»";
                    case "SET_BIRTH_ORDER":
                        return "Đặt thứ tự thủ công «" + name + "»: " + show("birthOrderOverride", before == null ? null : before.get("birthOrderOverride"))
                            + " → " + show("birthOrderOverride", after == null ? null : after.get("birthOrderOverride"));
                    default: return action + " «" + name + "»";
                }
            }
            case "PARENT_CHILD": {
                String rel = personName(any, "parentId", nameOf) + " là cha/mẹ của " + personName(any, "childId", nameOf);
                return ("DELETE".equals(action) ? "Gỡ quan hệ: " : "REVERT".equals(action) ? "Khôi phục quan hệ: " : "Thêm quan hệ: ") + rel;
            }
            case "UNION": {
                String pair = personName(any, "partnerAId", nameOf) + " – " + personName(any, "partnerBId", nameOf);
                return switch (action) {
                    case "DELETE" -> "Gỡ hôn phối: " + pair;
                    case "UPDATE" -> "Sửa hôn phối: " + pair;
                    case "REVERT" -> "Khôi phục hôn phối: " + pair;
                    default -> "Thêm hôn phối: " + pair;
                };
            }
            default: return action + " " + entityType;
        }
    }

    public static boolean revertible(String entityType, String action) {
        return switch (entityType) {
            case "PERSON" -> Set.of("CREATE", "UPDATE", "DELETE", "SET_BIRTH_ORDER").contains(action);
            case "PARENT_CHILD" -> Set.of("CREATE", "DELETE").contains(action);
            case "UNION" -> Set.of("CREATE", "UPDATE", "DELETE").contains(action);
            default -> false;
        };
    }

    private static String text(JsonNode n, String f) { return n == null || n.get(f) == null || n.get(f).isNull() ? "" : n.get(f).asText(); }

    private static String personName(JsonNode n, String f, Function<Long, String> nameOf) {
        if (n == null || !n.hasNonNull(f)) return "?";
        long id = n.get(f).asLong();
        String name = nameOf.apply(id);
        return name == null ? "#" + id : name;
    }

    private static String show(String field, JsonNode v) {
        if (v == null || v.isNull()) return "—";
        if (v.isBoolean()) return v.asBoolean() ? "Có" : "Không";
        String s = v.asText();
        if ("gender".equals(field)) return switch (s) { case "M" -> "Nam"; case "F" -> "Nữ"; default -> "Chưa rõ"; };
        return s.length() > 80 ? s.substring(0, 79) + "…" : s;
    }
}
```

```java
package vn.giapha.audit;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@Transactional(readOnly = true)
public class AuditQueryService {
    private static final String SELECT = """
        SELECT a.id, a.user_id, u.username, u.display_name, a.entity_type, a.entity_id, a.action,
               a.before_data::text, a.after_data::text, a.created_at
        FROM audit_log a LEFT JOIN app_user u ON u.id = a.user_id
        """;

    private final JdbcTemplate jdbc;
    private final ObjectMapper om;

    public AuditQueryService(JdbcTemplate jdbc, ObjectMapper om) {
        this.jdbc = jdbc;
        this.om = om;
    }

    private record Raw(long id, Long userId, String username, String displayName, String entityType, long entityId, String action,
                       JsonNode before, JsonNode after, java.time.Instant createdAt) {}

    public List<AuditEntry> forPerson(long personId, Long beforeId, int limit) {
        String id = String.valueOf(personId);
        List<Raw> rows = jdbc.query(SELECT + """
            WHERE (?::bigint IS NULL OR a.id < ?::bigint)
              AND ((a.entity_type = 'PERSON' AND a.entity_id = ?)
                OR (a.entity_type = 'PARENT_CHILD' AND (COALESCE(a.after_data, a.before_data) ->> 'parentId' = ? OR COALESCE(a.after_data, a.before_data) ->> 'childId' = ?))
                OR (a.entity_type = 'UNION' AND (COALESCE(a.after_data, a.before_data) ->> 'partnerAId' = ? OR COALESCE(a.after_data, a.before_data) ->> 'partnerBId' = ?)))
            ORDER BY a.id DESC LIMIT ?
            """, this::raw, beforeId, beforeId, personId, id, id, id, id, clamp(limit));
        return enrich(rows);
    }

    public List<AuditEntry> all(Long beforeId, int limit) {
        List<Raw> rows = jdbc.query(SELECT + " WHERE (?::bigint IS NULL OR a.id < ?::bigint) ORDER BY a.id DESC LIMIT ?",
            this::raw, beforeId, beforeId, clamp(limit));
        return enrich(rows);
    }

    private static int clamp(int limit) { return Math.max(1, Math.min(100, limit)); }

    private Raw raw(java.sql.ResultSet rs, int i) throws java.sql.SQLException {
        return new Raw(rs.getLong(1), (Long) rs.getObject(2), rs.getString(3), rs.getString(4), rs.getString(5), rs.getLong(6),
            rs.getString(7), parse(rs.getString(8)), parse(rs.getString(9)), rs.getTimestamp(10).toInstant());
    }

    private JsonNode parse(String s) {
        if (s == null) return null;
        try { return om.readTree(s); } catch (JsonProcessingException e) { return null; }
    }

    private List<AuditEntry> enrich(List<Raw> rows) {
        Set<Long> ids = new LinkedHashSet<>();
        for (Raw r : rows) {
            JsonNode n = r.after() != null ? r.after() : r.before();
            if (n == null) continue;
            for (String f : new String[]{"parentId", "childId", "partnerAId", "partnerBId"}) if (n.hasNonNull(f)) ids.add(n.get(f).asLong());
        }
        Map<Long, String> names = new HashMap<>();
        if (!ids.isEmpty()) {
            jdbc.query("SELECT id, full_name FROM person WHERE id = ANY(?)", rs -> { names.put(rs.getLong(1), rs.getString(2)); },
                (Object) ids.toArray(Long[]::new));
        }
        List<AuditEntry> out = new ArrayList<>();
        for (Raw r : rows) {
            out.add(new AuditEntry(r.id(), r.userId(), r.username(), r.displayName(), r.entityType(), r.entityId(), r.action(),
                r.before(), r.after(), r.createdAt(),
                AuditSummary.summary(r.entityType(), r.action(), r.before(), r.after(), names::get),
                AuditSummary.changes(r.entityType(), r.before(), r.after()),
                AuditSummary.revertible(r.entityType(), r.action())));
        }
        return out;
    }
}
```

```java
package vn.giapha.audit;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.giapha.auth.AuthUser;
import vn.giapha.clan.ClanMemberService;
import vn.giapha.common.ApiException;
import vn.giapha.person.PersonService;
import vn.giapha.relation.*;

@Service
public class RevertService {
    private record Row(long id, String type, long entityId, String action, JsonNode before, JsonNode after) {}

    private final JdbcTemplate jdbc;
    private final ObjectMapper om;
    private final PersonService persons;
    private final ParentChildService pcs;
    private final ParentChildRepository links;
    private final UnionService unions;
    private final UnionRepository unionRepo;
    private final ClanMemberService clans;
    private final AuditService audit;

    public RevertService(JdbcTemplate jdbc, ObjectMapper om, PersonService persons, ParentChildService pcs, ParentChildRepository links,
                         UnionService unions, UnionRepository unionRepo, ClanMemberService clans, AuditService audit) {
        this.jdbc = jdbc;
        this.om = om;
        this.persons = persons;
        this.pcs = pcs;
        this.links = links;
        this.unions = unions;
        this.unionRepo = unionRepo;
        this.clans = clans;
        this.audit = audit;
    }

    @Transactional
    public void revert(AuthUser me, long auditId) {
        Row r = load(auditId);
        if (!AuditSummary.revertible(r.type(), r.action())) throw notRevertible("Mục này không khôi phục được");
        switch (r.type()) {
            case "PERSON" -> revertPerson(me, r);
            case "PARENT_CHILD" -> revertLink(me, r);
            case "UNION" -> revertUnion(me, r);
            default -> throw notRevertible("Mục này không khôi phục được");
        }
    }

    private void revertPerson(AuthUser me, Row r) {
        switch (r.action()) {
            case "UPDATE", "SET_BIRTH_ORDER" -> persons.applyHistory(me, r.entityId(), r.before());
            case "CREATE" -> persons.softDelete(me, r.entityId(), "REVERT");
            case "DELETE" -> persons.restore(me, r.entityId());
            default -> throw notRevertible("Mục này không khôi phục được");
        }
    }

    private void revertLink(AuthUser me, Row r) {
        if ("CREATE".equals(r.action())) {
            ParentChild l = links.findById(r.entityId()).orElseThrow(() -> notRevertible("Quan hệ này đã bị gỡ"));
            var before = Snapshots.of(l);
            pcs.remove(l.getId());
            audit.record(me.id(), "PARENT_CHILD", l.getId(), "REVERT", before, null);
            clans.rebuildClansOf(List.of(l.getParentId(), l.getChildId()));
        } else {
            LinkSnapView s = om.convertValue(r.before(), LinkSnapView.class);
            Long unionId = s.unionId() != null && unionRepo.existsById(s.unionId()) ? s.unionId() : null;
            var res = pcs.add(s.parentId(), s.childId(), s.kind(), unionId);
            audit.record(me.id(), "PARENT_CHILD", res.link().getId(), "REVERT", null, Snapshots.of(res.link()));
            clans.rebuildClansOf(List.of(s.parentId(), s.childId()));
        }
    }

    private void revertUnion(AuthUser me, Row r) {
        switch (r.action()) {
            case "CREATE" -> {
                UnionRel u = unionRepo.findById(r.entityId()).orElseThrow(() -> notRevertible("Hôn phối này đã bị gỡ"));
                var before = Snapshots.of(u);
                unions.delete(u.getId());
                audit.record(me.id(), "UNION", u.getId(), "REVERT", before, null);
                clans.rebuildClansOf(List.of(u.getPartnerAId(), u.getPartnerBId()));
            }
            case "UPDATE" -> {
                UnionRel u = unionRepo.findById(r.entityId()).orElseThrow(() -> notRevertible("Hôn phối này đã bị gỡ"));
                var before = Snapshots.of(u);
                UnionSnapView s = om.convertValue(r.before(), UnionSnapView.class);
                unions.update(u.getId(), s.order(), s.status(), null, null);
                UnionRel fresh = unionRepo.findById(u.getId()).orElseThrow();
                fresh.setStartDate(s.startDate());
                fresh.setEndDate(s.endDate());
                unionRepo.saveAndFlush(fresh);
                audit.record(me.id(), "UNION", u.getId(), "REVERT", before, Snapshots.of(fresh));
            }
            case "DELETE" -> {
                UnionSnapView s = om.convertValue(r.before(), UnionSnapView.class);
                UnionRel created = unions.create(s.partnerAId(), s.partnerBId(), s.order(), s.status(), s.startDate(), s.endDate());
                audit.record(me.id(), "UNION", created.getId(), "REVERT", null, Snapshots.of(created));
                clans.rebuildClansOf(List.of(s.partnerAId(), s.partnerBId()));
            }
            default -> throw notRevertible("Mục này không khôi phục được");
        }
    }

    private Row load(long id) {
        List<Row> rows = jdbc.query("SELECT id, entity_type, entity_id, action, before_data::text, after_data::text FROM audit_log WHERE id = ?",
            (rs, i) -> new Row(rs.getLong(1), rs.getString(2), rs.getLong(3), rs.getString(4), parse(rs.getString(5)), parse(rs.getString(6))), id);
        if (rows.isEmpty()) throw new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND", "Không tìm thấy mục nhật ký");
        return rows.get(0);
    }

    private JsonNode parse(String s) {
        if (s == null) return null;
        try { return om.readTree(s); } catch (JsonProcessingException e) { return null; }
    }

    private ApiException notRevertible(String m) { return new ApiException(HttpStatus.CONFLICT, "NOT_REVERTIBLE", m); }

    private record LinkSnapView(Long parentId, Long childId, ParentKind kind, Long unionId) {}
    private record UnionSnapView(Long partnerAId, Long partnerBId, int order, UnionStatus status, java.time.LocalDate startDate, java.time.LocalDate endDate) {}
}
```

```tsx
import { useInfiniteQuery, useQueryClient } from '@tanstack/react-query'
import { App, Button, Empty, List, Popconfirm, Space, Spin, Table, Typography } from 'antd'
import { AUDIT_PAGE, auditApi, type AuditEntry } from '../../api/audit'
import { errorText } from '../../auth/errorText'
import { formatDateTime } from '../../lib/format'

export default function AuditTab({ personId, canRevert, onReverted }: { personId: number; canRevert: boolean; onReverted(): void }) {
  const { message } = App.useApp()
  const qc = useQueryClient()
  const q = useInfiniteQuery({
    queryKey: ['audit', personId],
    queryFn: ({ pageParam }) => auditApi.list(personId, pageParam, AUDIT_PAGE),
    initialPageParam: undefined as number | undefined,
    getNextPageParam: (last: AuditEntry[]) => (last.length === AUDIT_PAGE ? last[last.length - 1].id : undefined),
  })

    try {
      await auditApi.revert(id)
      message.success('Đã khôi phục')
      await qc.invalidateQueries({ queryKey: ['audit', personId] })
      onReverted()
    } catch (e) {
      message.error(errorText((e as { code?: string }).code))
    }
  }

  if (q.isLoading) return <Spin />
  const items = q.data?.pages.flat() ?? []
  if (!items.length) return <Empty description="Chưa có lịch sử sửa" />
  return (

        <List.Item key={e.id} extra={canRevert && e.revertible ? (
          <Popconfirm title="Khôi phục về trạng thái trước lần thay đổi này?" okText="Khôi phục bản này" cancelText="Huỷ" onConfirm={() => revert(e.id)}>

                columns={[{ title: 'Trường', dataIndex: 'label' }, { title: 'Trước', dataIndex: 'from' }, { title: 'Sau', dataIndex: 'to' }]} />
            )}
          </Space>
        </List.Item>
      )} />
      {q.hasNextPage && <Button block loading={q.isFetchingNextPage} onClick={() => q.fetchNextPage()}>Tải thêm</Button>}
    </div>
  )
}
```
